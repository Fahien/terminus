{#- Generates responsive images from a given high-resolution source image -#} {#- Tera V2 components are hygienic, so `page` and `config` are passed in. -#} {%- component ui.responsive_image(src, alt, page, config, caption="", lazy=true) -%} {%- set source_image_path = page.colocated_path ~ src -%} {%- set metadata = get_image_metadata(path=source_image_path) -%} {%- set image_candidates = [] -%} {%- for width in config.extra.responsive_images.widths -%} {%- if metadata.width >= width -%} {%- set resized_image = resize_image(path=source_image_path, width=width, op="fit_width", quality=80) -%} {%- set_global image_candidates = [...image_candidates, resized_image.url ~ " " ~ width ~ "w"] -%} {%- endif -%} {%- endfor -%} {%- set fallback_width = config.extra.responsive_images.fallback_width -%} {%- set fallback_image = resize_image(path=source_image_path, width=fallback_width, height=fallback_width, op="fit") -%} {%- set fallback_height = (fallback_image.width / (metadata.width / metadata.height)) | round(method="floor") | int -%} {#- Browser hint: main content width will max out at 784px in practice -#} {%- if config.extra.layout == "full-width" -%} {%- set sizes = "(min-width: 700px) calc(82vw + 46px), calc(100vw - 40px)" -%} {%- else -%} {%- set sizes = "(min-width: 920px) 784px, (min-width: 700px) calc(82vw + 46px), calc(100vw - 40px)" -%} {%- endif -%} {%- if lazy -%} {%- set loading = ' loading="lazy"' -%} {%- else -%} {%- set loading = ' fetchpriority="high"' -%} {%- endif -%} {#- The tag is emitted on a single line so Markdown treats it as an HTML block -#} {#- rather than wrapping the image in a paragraph. -#} {%- if caption %} {%- else %} {%- endif %} {%- endcomponent ui.responsive_image -%}
